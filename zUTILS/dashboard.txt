        async function cargarDashboard() {
            // 1. Revisamos si el usuario trae su "gafete" (Token)
            const token = localStorage.getItem('token');
            const esEstudiante = localStorage.getItem('es_estudiante'); // Leemos el rol
            
            if (!token) {
                window.location.href = "index.html";
                return;
            }

            // APLICAMOS LA MAGIA DE LOS ROLES (MODO SEGURO)
            if (esEstudiante === 'true') {
                // Ocultamos las cosas de admin al estudiante (SOLO SI EXISTEN)
                const btnCrud = document.getElementById('btnCrudTarjetas');
                if (btnCrud) btnCrud.style.display = 'none';
                
                const btnVerificar = document.getElementById('btnVerificarEstatus');
                if (btnVerificar) btnVerificar.style.display = 'none';
                
                const btnDocs = document.getElementById('btnVerificarDocs');
                if (btnDocs) btnDocs.style.display = 'none';
            } else { 
                // <--- AQUÍ EMPIEZA EL BLOQUE QUE MENCIONAS --->
                // Si es admin, ocultamos Billetera, recargas, historial y RENOVACIÓN
                const btnHist = document.getElementById('btnHistorial');
                if (btnHist) btnHist.style.display = 'none';
                
                const btnRecarga = document.getElementById('btnRecarga');
                if (btnRecarga) btnRecarga.style.display = 'none';
                
                const panelBill = document.getElementById('panelBilletera');
                if (panelBill) panelBill.style.display = 'none';

                // NUEVO BOTÓN OCULTO
                const btnReno = document.getElementById('btnRenovacion');
                if (btnReno) btnReno.style.display = 'none';

                const badge = document.getElementById('badgeVerificacion');
                if (badge) badge.style.display = 'none';
            }
            try {
                // 2. Le preguntamos al backend de quién es este token
                    const token = localStorage.getItem('token');
                    const response = await fetch('http://127.0.0.1:8000/api/mis-tarjetas', {
                        headers: { 'Authorization': `Bearer ${token}` }
                });

                if (response.ok) {
                    const datosUsuario = await response.json();
                    
                    // 3. Modificamos el HTML para saludarlo por su nombre real
                    // Busca en tu HTML dónde dice "Hola, Usuario" o similar y ponle un id="nombreUsuario"
                    const elementoNombre = document.getElementById('nombreUsuario');
                    if (elementoNombre) {
                        // Solo mostramos el primer nombre para que se vea estético
                        const primerNombre = datosUsuario.nombre.split(' ')[0]; 
                        elementoNombre.innerText = `Hola, ${primerNombre}`;
                    }
                    cargarTarjetasBilletera();
                } else {
                    // Si el token es falso o ya expiró (pasó 1 hora)
                    cerrarSesion();
                }
            } catch (error) {
                console.error("Error al conectar con el servidor", error);
            }
        }

        // Función para destruir el token y salir
        function cerrarSesion() {
            localStorage.removeItem('token');
            window.location.href = "index.html";
        }

        // Ejecutamos la protección apenas se abre la página
        cargarDashboard();

        // NUEVA FUNCIÓN: Cargar tarjetas en el selector interactivo
        async function cargarTarjetasBilletera() {
            try {
                const token = localStorage.getItem('token');
                const response = await fetch('http://127.0.0.1:8000/api/mis-tarjetas', {
                    headers: { 'Authorization': `Bearer ${token}` }
                });

                if (response.ok) {
                    const tarjetas = await response.json();
                    
                    // Usamos TUS IDs exactos del HTML
                    const select = document.getElementById('selectorTarjetas'); 

                    select.innerHTML = ''; // Borramos el texto de "Cargando tarjetas..."

                    if (tarjetas.length === 0) {
                        document.getElementById('saldoMostrado').innerText = "$ 0.00";
                        document.getElementById('estatusMostrado').innerText = "Sin tarjeta asignada";
                        select.innerHTML = '<option value="">Sin tarjeta asignada</option>';
                        return; // Detenemos aquí si no hay tarjetas
                    }

                    // Llenamos el select con la tarjeta del estudiante
                    tarjetas.forEach(t => {
                        const option = document.createElement('option');
                        option.value = t.tarjeta_id;
                        option.text = `Tarjeta terminación: ${t.numero_tarjeta.slice(-4)}`;
                        option.dataset.saldo = t.saldo;
                        option.dataset.estatus = t.estatus;
                        select.appendChild(option);
                    });

                    // Actualizamos los números grandes en pantalla
                    document.getElementById('saldoMostrado').innerText = `$ ${tarjetas[0].saldo.toFixed(2)}`;
                    
                    const spanEstatus = document.getElementById('estatusMostrado');
                    spanEstatus.innerText = tarjetas[0].estatus;

                    // Le damos color al estatus
                    if(tarjetas[0].estatus === "Activa") {
                        spanEstatus.style.backgroundColor = "rgba(46, 204, 113, 0.2)";
                        spanEstatus.style.color = "#2ecc71";
                    } else {
                        spanEstatus.style.backgroundColor = "rgba(231, 76, 60, 0.2)";
                        spanEstatus.style.color = "#e74c3c";
                    }

                    // Evento por si en el futuro tienen más de una tarjeta
                    select.addEventListener('change', (e) => {
                        const selectedOption = e.target.options[e.target.selectedIndex];
                        document.getElementById('saldoMostrado').innerText = `$ ${parseFloat(selectedOption.dataset.saldo).toFixed(2)}`;
                        spanEstatus.innerText = selectedOption.dataset.estatus;
                    });
                }
            } catch (error) {
                console.error("Error cargando la billetera:", error);
            }
        }
    </script>
